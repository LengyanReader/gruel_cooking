This mirrors the static pages so the two readers behave identically. #} {% macro t(zh, en) -%} {%- if lang == 'en' -%}{{ en }} {%- elif lang == 'zh' -%}{{ zh }} {%- else -%}{{ zh }}{{ en }}{%- endif -%} {%- endmacro %} {# Plain-text variant, for attributes (aria-label, title). The `t` macro deliberately emits markup so the stylesheet can switch halves in dual mode; that is invalid inside an attribute value, so attributes use this one instead. #} {% macro tplain(zh, en) -%} {%- if lang == 'en' -%}{{ en }} {%- elif lang == 'zh' -%}{{ zh }} {%- else -%}{{ zh }} / {{ en }}{%- endif -%} {%- endmacro %}
